// ================================================================
// 杀敌惩罚 — 每个敌人扣 100 火苗，火苗归零警告
// ================================================================

_VM_BATTLE_START {
    VM_ClearPlants(-1, -1)
    VM_SetFlame(500)
    VM_ShowNotice("击杀敌人会消耗火苗！")
}

_VM_ENEMY_KILLED {
    flame = VM_GetFlame()
    penalty = flame - 100
    VM_SetFlame(penalty)
    new_flame = VM_GetFlame()
    VM_ShellPrint("扣除100，剩余:", new_flame)

    if (new_flame <= 0) {
        VM_SetFlame(0)
        VM_ShowNotice("火苗耗尽！")
    }
}

_VM_WAVE_START {
    wave = VM_GetWave()
    if (wave == 0) {
        VM_SpawnEnemy("normal_mouse", 2, 100)
        VM_SpawnEnemy("normal_mouse", 3, 100)
        VM_SpawnEnemy("football_fan_mouse", 4, 250)
    }
    if (wave == 1) {
        VM_SpawnEnemy("iron_pan_mouse", 2, 660)
        VM_SpawnEnemy("skateboard_mouse", 1, 170)
        VM_SpawnEnemy("machine_mouse", 3, 170)
    }
    if (wave == 2) {
        VM_SpawnBoss("mario_mouse", 2, 9000)
        VM_ShowNotice("BOSS 来了！")
    }
}
