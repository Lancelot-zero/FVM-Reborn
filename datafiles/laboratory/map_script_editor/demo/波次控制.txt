// ================================================================
// Demo 1 — 简单波次控制
// 每波开始时刷敌人，第5波刷BOSS，最后波胜利提示
// ================================================================

_VM_WAVE_START {
    wave = VM_GetWave()

    if (wave == 0) {
        VM_ShowNotice("战斗开始！")
        VM_SpawnEnemy("normal_mouse", 2, 100)
        VM_SpawnEnemy("normal_mouse", 3, 100)
    }
    if (wave == 1) {
        VM_ShowNotice("第二波")
        VM_SpawnEnemy("football_fan_mouse", 1, 200)
        VM_SpawnEnemy("skateboard_mouse", 4, 150)
    }
    if (wave == 2) {
        VM_ShowNotice("第三波")
        VM_SpawnEnemy("iron_pan_mouse", 2, 500)
        VM_SpawnEnemy("arson_mouse", 3, 400)
    }
    if (wave == 3) {
        VM_ShowNotice("第四波！")
        VM_SpawnEnemy("machine_mouse", 1, 300)
        VM_SpawnEnemy("flute_mouse", 3, 800)
        VM_SpawnEnemy("ninja_mouse", 4, 400)
    }
    if (wave == 4) {
        VM_ShowNotice("BOSS 登场！！")
        VM_SpawnBoss("arno", 2, 5000)
    }
}

_VM_ENEMY_KILLED {
    enemy = VM_GetLastKilledEnemy()
    VM_ShellPrint("击杀:", enemy)
}
