// ================================================================
// 障碍跑酷 — 全图水+移动平台，只有平台和终点能种卡
// ================================================================

_VM_ROOM_READY_ENTRY {
    VM_SetMaxSlots(5)
}

_VM_BATTLE_START {
    VM_ClearPlants(-1, -1)
    VM_SetTerrain(-1, -1, "water")
    VM_SetFlame(1000)

    // 一排移动平台，从左到右
    VM_CreatePlatform(0, 2, 4, 3, 0, 2, 480, "spr_fennel_raft_platform_daytime")
    VM_CreatePlatform(3, 0, 5, 1, 1, 2, 480, "spr_lilac_rainbow_platform_night_1")
    VM_CreatePlatform(5, 2, 4, 3, 0, 1, 480, "spr_lilac_rainbow_platform_night_2")
    VM_CreatePlatform(7, 0, 5, 1, 1, 1, 480, "spr_lilac_rainbow_platform_night_6")
    VM_CreatePlatform(8, 2, 4, 3, 0, 1, 480, "spr_lilac_rainbow_platform_night_7")

    // 终点安全区
    VM_SetTerrain(9, -1, "normal")
}

_VM_WAVE_START {
    wave = VM_GetWave()
    if (wave == 0) {
        VM_SpawnEnemy("normal_mouse", 2, 50)
        VM_ShowNotice("小心别掉水里！")
    }
    if (wave == 1) {
        VM_SpawnEnemy("skateboard_mouse", 1, 150)
        VM_SpawnEnemy("glider_mouse", 4, 120)
    }
    if (wave == 2) {
        VM_SpawnEnemy("diver_mouse", 3, 200)
        VM_SpawnBoss("blonde_mary", 2, 8000)
        VM_ShowNotice("BOSS 来了！")
    }
}
