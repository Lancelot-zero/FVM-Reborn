// ================================================================
// Demo 7 — 开局限制：禁卡、限槽、限星
// 只允许 5 张卡 + 最高 8 星，禁用产火和炸弹
// ================================================================

_VM_ROOM_READY_ENTRY {
    // 最多带 5 张卡
    VM_SetMaxSlots(5)

    // 最高 8 星
    VM_SetCardLevelCap(8)

    // 禁用产火卡
    VM_BanCard("small_fire")
    VM_BanCard("large_fire")
    VM_BanCard("goblet_lamp")
    VM_BanCard("firework_dragon")

    // 禁用炸弹
    VM_BanCard("coke_bomb")
    VM_BanCard("kettle_bomb")
    VM_BanCard("wine_bottle_bomb")
    VM_BanCard("ice_bucket_bomb")

    // 禁用木盘子
    VM_BanCard("wooden_plate")
}

_VM_BATTLE_START {
    VM_ClearPlants(-1, -1)
    VM_SetTerrain(-1, -1, "obstacle")

    // 只有中间3行可种植
    VM_SetTerrain(-1, 2, "normal")
    VM_SetTerrain(-1, 3, "normal")
    VM_SetTerrain(-1, 4, "normal")

    // 开局只有 500 火苗
    VM_SetFlame(500)

    VM_ShowNotice("禁用产火卡+炸弹，只有5槽8星")
}

_VM_WAVE_START {
    wave = VM_GetWave()
    if (wave == 0) {
        VM_SpawnEnemy("normal_mouse", 2, 100)
        VM_SpawnEnemy("normal_mouse", 3, 100)
    }
    if (wave == 1) {
        VM_SpawnEnemy("football_fan_mouse", 2, 280)
        VM_SpawnEnemy("landlady_mouse", 3, 200)
        VM_SpawnEnemy("skateboard_mouse", 4, 170)
    }
    if (wave == 2) {
        VM_SpawnEnemy("iron_pan_mouse", 2, 660)
        VM_SpawnEnemy("arson_mouse", 3, 600)
        VM_SpawnEnemy("non_mainstream_mouse", 1, 500)
        VM_ShowNotice("第三波！")
    }
    if (wave == 3) {
        VM_SpawnBoss("blonde_mary", 2, 12000)
        VM_ShowNotice("最终 BOSS！")
    }
}
