_VM_ROOM_READY_ENTRY {
    VM_ShowNotice("欢迎来到实验室！")
    VM_ShellPrint("[准备室] 规则已设置")
}

_VM_BATTLE_START {
    VM_ClearPlants(-1, -1)
    VM_SetFlame(1000)
    VM_ShowNotice("战斗开始，祝你好运！")
    VM_ShellPrint("[战斗开始] 地图初始化完成")
}

_VM_WAVE_START {
    wave = VM_GetWave()
    if (wave == 0) {
        VM_ShowNotice("第一波：平民鼠来袭")
        VM_ShellPrint("[波次] 第0波开始")
        VM_SpawnEnemy("normal_mouse", 2, 100)
        VM_SpawnEnemy("normal_mouse", 3, 100)
    }
    if (wave == 1) {
        VM_ShowNotice("第二波：球迷鼠和滑板鼠")
        VM_ShellPrint("[波次] 第1波开始")
        VM_SpawnEnemy("football_fan_mouse", 1, 280)
        VM_SpawnEnemy("skateboard_mouse", 4, 170)
    }
    if (wave == 2) {
        VM_ShowNotice("第三波：铁锅鼠来了！")
        VM_ShellPrint("[波次] 第2波开始")
        VM_SpawnEnemy("iron_pan_mouse", 2, 660)
    }
    if (wave == 3) {
        VM_ShowNotice("最终波：BOSS 阿诺登场！")
        VM_ShellPrint("[波次] 第3波 — BOSS战")
        VM_SpawnBoss("arno", 2, 8000)
    }
}
