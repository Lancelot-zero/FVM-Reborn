// ================================================================
// 平台循环 — 三个平台各自不同循环轨迹
// ================================================================

_VM_BATTLE_START {
    VM_ClearPlants(-1, -1)
    VM_SetTerrain(-1, -1, "obstacle")
    VM_SetFlame(2000)

    // 平台 A：大平台 4x5，上下小步循环
    a_cnt = 0
    a = VM_CreatePlatform(0, 1, 4, 5, 0, 0, 0, "spr_fennel_raft_platform_daytime")

    // 平台 B：5x1，左右大步循环
    b_cnt = 0
    b = VM_CreatePlatform(6, 2, 5, 1, 0, 0, 0, "spr_lilac_rainbow_platform_night_1")

    // 平台 C：4x5，走8字（右→下→左→上）
    c_cnt = 0
    c = VM_CreatePlatform(3, 4, 4, 5, 0, 0, 0, "spr_lilac_rainbow_platform_night_6")
}

_VM_PLATFORM_IDLE_END {
    plat = VM_GetLastIdlePlatform()

    if (plat == a) {
        a_cnt = (a_cnt + 1) % 2
        if (a_cnt == 0) {
            VM_SetPlatformParams(a, 0, 1, 120, 1)
        } else {
            VM_SetPlatformParams(a, 0, 1, 120, -1)
        }
    }

    if (plat == b) {
        b_cnt = (b_cnt + 1) % 2
        if (b_cnt == 0) {
            VM_SetPlatformParams(b, 1, 3, 480, 1)
        } else {
            VM_SetPlatformParams(b, 1, 3, 480, -1)
        }
    }

    if (plat == c) {
        c_cnt = (c_cnt + 1) % 4
        if (c_cnt == 0) {
            VM_SetPlatformParams(c, 1, 2, 480, 1)
        }
        if (c_cnt == 1) {
            VM_SetPlatformParams(c, 0, 2, 480, 1)
        }
        if (c_cnt == 2) {
            VM_SetPlatformParams(c, 1, 2, 480, -1)
        }
        if (c_cnt == 3) {
            VM_SetPlatformParams(c, 0, 2, 480, -1)
        }
    }
}
